    \begin{tikzpicture}[

        % A framework for adaptive sorting
        % by O. Pertersson and A. Moffat
        %
        % Max=Dis equivalence comes from:
        % NeatSort - A practical adaptive algorithm
        % by M. La Rocca and D. Cantone
        \node[ellipse] (reg) {$\mathit{Reg}$};
        \node[ellipse] (loc) [below of=reg] {$\mathit{Loc}$};
        \node[ellipse] (sms) [right=of loc] {$\mathit{SMS}$};
        \node[ellipse] (osc) [below of=loc] {$\mathit{Osc}$};
        \node[ellipse] (hist) [left=of loc] {$\mathit{Hist}$};
        \node[circle] (blank) [below=of osc] {};
        \node[ellipse] (enc) [below of=sms] {$\mathit{Enc}$};
        \node[ellipse] (inv) [below of=blank] {$\mathit{Inv}~$$\equiv$$~\mathit{Spear}$};
        \node[ellipse] (rem) [left=of inv] {$\mathit{Rem}$};
        \node[ellipse] (exc) [below of=rem] {$\mathit{Exc}~$$\equiv$$~\mathit{Ham}$};
        \node[ellipse] (block) [above=of rem] {$\mathit{Block}$};
        \node[ellipse] (max) [below of=inv] {$\mathit{Max}~$$\equiv$$~\mathit{Dis}~$};
        \node[ellipse] (sus) [right=of inv] {$\mathit{SUS}$};
        \node[ellipse] (runs) [right=of max] {$\mathit{Runs}$};
        \node[ellipse] (m01) [below of=max] {$m_{01}$};
        \node[ellipse] (m0) [below of=m01] {$m_{0}$};

              (exc) -- (m01)
              (max) -- (m01)
              (runs) -- (m01)
              (m01) -- (m0);

        \node[ellipse] (amp) [right of=blank2] {$\mathit{Amp}$};
